4.HOTEL CLASS
import java.util.*;

public class Hotel {

    private List<Room> rooms = new ArrayList<>();
    private List<Reservation> reservations = new ArrayList<>();

    public void addRoom(Room room) {
        rooms.add(room);
    }

    public void displayAvailableRooms() {

        System.out.println("\nAvailable Rooms:");

        for(Room room : rooms) {
            if(room.isAvailable()) {
                System.out.println(
                    room.getRoomNumber() + " | " +
                    room.getCategory() + " | ₹" +
                    room.getPrice()
                );
            }
        }
    }

    public Reservation bookRoom(
            int bookingId,
            Customer customer,
            int roomNumber,
            String checkIn,
            String checkOut) {

        for(Room room : rooms) {

            if(room.getRoomNumber() == roomNumber
                    && room.isAvailable()) {

                room.setAvailable(false);

                Reservation reservation =
                        new Reservation(
                                bookingId,
                                customer,
                                room,
                                java.time.LocalDate.parse(checkIn),
                                java.time.LocalDate.parse(checkOut)
                        );

                reservations.add(reservation);

                return reservation;
            }
        }
        return null;
    }

    public boolean cancelBooking(int bookingId) {

        Iterator<Reservation> iterator =
                reservations.iterator();

        while(iterator.hasNext()) {

            Reservation reservation = iterator.next();

            if(reservation.getBookingId() == bookingId) {

                reservation.getRoom()
                        .setAvailable(true);

                iterator.remove();

                return true;
            }
        }
        return false;
    }
}

5.FILE STORAGE
import java.io.FileWriter;
import java.io.IOException;

public class FileManager {

    public static void saveBooking(
            Reservation reservation)
            throws IOException {

        FileWriter writer =
                new FileWriter(
                    "bookings.txt",
                    true);

        writer.write(
            reservation.getBookingId() + "," +
            reservation.getCustomer().getName() + "," +
            reservation.getRoom().getRoomNumber() + "\n"
        );

        writer.close();
    }
}
